if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "lumatape_capture requires Windows x64, MSVC and Windows SDK 10.0.26100 or newer")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "lumatape_capture supports x64 only")
endif()
if(CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION AND CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION VERSION_LESS "10.0.26100.0")
  message(FATAL_ERROR "Select Windows SDK 10.0.26100.0 or newer for its modern C++/WinRT headers (runtime minimum remains Windows 10 build 19041)")
endif()

add_library(lumatape_capture SHARED capture.cpp capture.h)
target_compile_features(lumatape_capture PRIVATE cxx_std_20)
target_compile_definitions(lumatape_capture PRIVATE VHS_CAPTURE_BUILD WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_compile_options(lumatape_capture PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(lumatape_capture PRIVATE d3d11 dxgi dwmapi opengl32 runtimeobject windowsapp user32)
# Keep the optional bridge self-contained: no additional VC runtime DLL needed.
set_property(TARGET lumatape_capture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
install(TARGETS lumatape_capture RUNTIME DESTINATION .)

if(BUILD_TESTING)
  add_executable(lumatape_capture_abi_smoke abi_smoke.cpp)
  target_compile_features(lumatape_capture_abi_smoke PRIVATE cxx_std_20)
  target_link_libraries(lumatape_capture_abi_smoke PRIVATE lumatape_capture)
  set_property(TARGET lumatape_capture_abi_smoke PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_test(NAME capture_abi_smoke COMMAND lumatape_capture_abi_smoke)
  install(TARGETS lumatape_capture_abi_smoke RUNTIME DESTINATION .)
endif()
